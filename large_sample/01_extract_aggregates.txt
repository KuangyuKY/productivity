/* ============================================================
   01_extract_aggregates —— 5 张聚合查询（合并版）
   表1/2/5：样本口径，各一条查询，单独跑 + 导出对应 CSV。
   表3/4：全量口径，改为「逐源表预聚合到 staging 永久表，再出结果」，
          降低 tempdb 峰值；且每张 city 表用完即 DROP staging，避免撑满 GX17。
   ★ 关键教训：staging 表建在当前库(GX17)里，占的是 GX17 磁盘。
     表3 的 cb_stage 若不删、一直占着，表4 的 cs_stage 叠上去就会撑满 GX17(1101)。
     所以表4 开头先把 cb_stage 及 00 的中间表清掉腾空间。
   样本表：dbo.tmp_sample_cid2（由 00_build_sample 生成，永久表，重启不丢）
   数据源：2017 GX17.dbo.GX1701/1702/1703 ｜ 2018 GX.dbo.gaoxiao1801 + GX02.dbo.GX02
   注意：数量用 TRY_CAST 转 float；红冲负值由 SUM 自动对冲为净额；
        「聚合消除了 Null 值」是无害警告，忽略。
   ============================================================ */


/* ############################################################
   表1：企业购买表（样本）→ 导出 firm_buy.csv
   —— 单独跑，导出结果
   ############################################################ */
SELECT 购方企业ID, 项目代码, 单位, [year],
       SUM(开票金额) AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM (
    SELECT 购方企业ID, 项目代码, 单位, 开票金额, 数量, 2017 AS [year] FROM GX17.dbo.GX1701
    UNION ALL SELECT 购方企业ID, 项目代码, 单位, 开票金额, 数量, 2017 FROM GX17.dbo.GX1702
    UNION ALL SELECT 购方企业ID, 项目代码, 单位, 开票金额, 数量, 2017 FROM GX17.dbo.GX1703
    UNION ALL SELECT 购方企业ID, 项目代码, 单位, 开票金额, 数量, 2018 FROM GX.dbo.gaoxiao1801
    UNION ALL SELECT 购方企业ID, 项目代码, 单位, 开票金额, 数量, 2018 FROM GX02.dbo.GX02
) a
WHERE EXISTS (SELECT 1 FROM dbo.tmp_sample_cid2 s WHERE s.cid = a.购方企业ID)
GROUP BY 购方企业ID, 项目代码, 单位, [year];

   表4：地区销售表（全量）→ 导出 city_sell.csv
   与表3 完全相同的做法（逐源表预聚合到 staging，再出结果），
   唯一区别：★ 显式 USE GX，把 cs_stage 建在 GX 库。

/* ############################################################
   表5：企业-地区对照表（样本，买方侧）→ 导出 firm_city.csv
   —— 单独跑，导出结果
   ############################################################ */
SELECT 企业ID, [year], 地区, COUNT(*) AS cnt
FROM (
    SELECT 购方企业ID AS 企业ID, 购方地区 AS 地区, 2017 AS [year] FROM GX17.dbo.GX1701
    UNION ALL SELECT 购方企业ID, 购方地区, 2017 FROM GX17.dbo.GX1702
    UNION ALL SELECT 购方企业ID, 购方地区, 2017 FROM GX17.dbo.GX1703
    UNION ALL SELECT 购方企业ID, 购方地区, 2018 FROM GX.dbo.gaoxiao1801
    UNION ALL SELECT 购方企业ID, 购方地区, 2018 FROM GX02.dbo.GX02
) a
WHERE 地区 IS NOT NULL
  AND EXISTS (SELECT 1 FROM dbo.tmp_sample_cid2 s WHERE s.cid = a.企业ID)
GROUP BY 企业ID, [year], 地区;
