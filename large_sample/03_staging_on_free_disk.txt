/* ============================================================
   03_staging_on_free_disk —— 把 staging 放到空闲磁盘的独立数据库
   目的：表3/表4 的 staging 不再建进 GX17（那块盘满了，报 1101），
        改建到空盘上的新库 WORKDB，GX17 完全不写入。
   ★ 先跑 02_check_space.txt，确认哪块盘空，把下面的 D:\ 改成那块盘。
   ============================================================ */


/* ============ 步骤 1：在空盘上建工作库 WORKDB（只需一次） ============ */
-- ★★ 把 D:\SQLWork 改成你 VM 上"空闲那块盘"的实际路径；目录需已存在或 SQL 有权限创建
IF DB_ID('WORKDB') IS NULL
CREATE DATABASE WORKDB
ON PRIMARY ( NAME = WORKDB_dat,
             FILENAME = 'D:\SQLWork\WORKDB.mdf',
             SIZE = 2GB, FILEGROWTH = 1GB )
LOG ON      ( NAME = WORKDB_log,
             FILENAME = 'D:\SQLWork\WORKDB.ldf',
             SIZE = 512MB, FILEGROWTH = 256MB );
GO

-- 简单恢复模式：批量插入时日志不会无限膨胀（重要）
ALTER DATABASE WORKDB SET RECOVERY SIMPLE;
GO


/* ============ 步骤 2：清掉之前建在 GX17 里的中间表，把 GX17 空间还回去 ============ */
USE GX17;
IF OBJECT_ID('dbo.cb_stage') IS NOT NULL DROP TABLE dbo.cb_stage;
IF OBJECT_ID('dbo.cs_stage') IS NOT NULL DROP TABLE dbo.cs_stage;
IF OBJECT_ID('dbo.s_both17') IS NOT NULL DROP TABLE dbo.s_both17;
IF OBJECT_ID('dbo.s_both18') IS NOT NULL DROP TABLE dbo.s_both18;
IF OBJECT_ID('dbo.s_pool')   IS NOT NULL DROP TABLE dbo.s_pool;
-- ★ 保留 dbo.tmp_sample_cid2（10 万样本名单，导数还要用）
GO


/* ############################################################
   表4：city_sell —— staging 建在 WORKDB（不碰 GX17）
   ############################################################ */

/* --- 4.1 建 cs_stage（整段一次跑；慢，耐心等） --- */
USE WORKDB;
IF OBJECT_ID('dbo.cs_stage') IS NOT NULL DROP TABLE dbo.cs_stage;

SELECT 销方地区, 销方企业ID, 项目代码, 单位, CAST(2017 AS int) AS [year],
       SUM(开票金额) AS 金额, SUM(TRY_CAST(数量 AS float)) AS 数量
INTO dbo.cs_stage
FROM GX17.dbo.GX1701 WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 销方企业ID, 项目代码, 单位;

INSERT INTO dbo.cs_stage
SELECT 销方地区, 销方企业ID, 项目代码, 单位, 2017, SUM(开票金额), SUM(TRY_CAST(数量 AS float))
FROM GX17.dbo.GX1702 WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 销方企业ID, 项目代码, 单位;

INSERT INTO dbo.cs_stage
SELECT 销方地区, 销方企业ID, 项目代码, 单位, 2017, SUM(开票金额), SUM(TRY_CAST(数量 AS float))
FROM GX17.dbo.GX1703 WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 销方企业ID, 项目代码, 单位;

INSERT INTO dbo.cs_stage
SELECT 销方地区, 销方企业ID, 项目代码, 单位, 2018, SUM(开票金额), SUM(TRY_CAST(数量 AS float))
FROM GX.dbo.gaoxiao1801 WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 销方企业ID, 项目代码, 单位;

INSERT INTO dbo.cs_stage
SELECT 销方地区, 销方企业ID, 项目代码, 单位, 2018, SUM(开票金额), SUM(TRY_CAST(数量 AS float))
FROM GX02.dbo.GX02 WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 销方企业ID, 项目代码, 单位;
GO

/* --- 4.2 出结果（单独跑）→ 导出 city_sell.csv --- */
USE WORKDB;
SELECT 销方地区, 项目代码, 单位, [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数,
       SUM(金额) AS 金额合计, SUM(数量) AS 数量合计
FROM dbo.cs_stage
GROUP BY 销方地区, 项目代码, 单位, [year];
