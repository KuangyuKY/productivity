/* ############################################################
   第 0 步：清理我们自己建的中间表（必须先做！）
   cs_stage 是 3.4 亿行，正是它把 D 盘写满、影响共享服务器。
   表3 city_buy.csv 已导出成功，cb_stage 也可删。
   DROP 很快，即使盘满也能执行。

USE GX17;

IF OBJECT_ID('dbo.cb_stage') IS NOT NULL DROP TABLE dbo.cb_stage;
IF OBJECT_ID('dbo.s_both17') IS NOT NULL DROP TABLE dbo.s_both17;

/* --- 第 0.1 步：确认空间已释放（GX17 空闲MB 应明显变大） --- */
USE GX17;
SELECT name AS 逻辑名,
       CAST(size/128.0 AS decimal(18,1))                                    AS 文件MB,
       CAST(FILEPROPERTY(name,'SpaceUsed')/128.0 AS decimal(18,1))          AS 已用MB,
       CAST((size - FILEPROPERTY(name,'SpaceUsed'))/128.0 AS decimal(18,1)) AS 空闲MB
FROM sys.database_files;


/* --- 第 0.2 步：列出所有省份前2位 + 各自数据量，决定要跑哪些片、以及哪些省需要再细拆 ---
   （这条本身也要排序，若报 tempdb 满，就跳过它，直接按下面的常见省份列表逐个跑） */
SELECT LEFT(销方地区,2) AS 省, COUNT(*) AS 行数
FROM GX17.dbo.GX1701
WHERE 销方地区 IS NOT NULL
GROUP BY LEFT(销方地区,2)
ORDER BY 行数 DESC;


/* ############################################################
   第 1 步：2017 年，逐省跑 —— 每跑完一个省导出一个 CSV
   用法：把下面 @省 那行的 '11' 改成目标省份，跑，导出
        city_sell_2017_11.csv，再换下一个省。
   ############################################################ */
SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数,
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM (
    SELECT 销方地区, 销方企业ID, 项目代码, 单位, 开票金额, 数量
      FROM GX17.dbo.GX1701 WHERE LEFT(销方地区,2) = '11'
    UNION ALL
    SELECT 销方地区, 销方企业ID, 项目代码, 单位, 开票金额, 数量
      FROM GX17.dbo.GX1702 WHERE LEFT(销方地区,2) = '11'
    UNION ALL
    SELECT 销方地区, 销方企业ID, 项目代码, 单位, 开票金额, 数量
      FROM GX17.dbo.GX1703 WHERE LEFT(销方地区,2) = '11'
) a
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;
-- → 导出 city_sell_2017_11.csv


/* ############################################################
   第 2 步：2018 年，逐省跑 —— 同样每省导出一个 CSV
   ############################################################ */
SELECT 销方地区, 项目代码, 单位, 2018 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数,
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM (
    SELECT 销方地区, 销方企业ID, 项目代码, 单位, 开票金额, 数量
      FROM GX.dbo.gaoxiao1801 WHERE LEFT(销方地区,2) = '11'
    UNION ALL
    SELECT 销方地区, 销方企业ID, 项目代码, 单位, 开票金额, 数量
      FROM GX02.dbo.GX02     WHERE LEFT(销方地区,2) = '11'
) a
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;
-- → 导出 city_sell_2018_11.csv
