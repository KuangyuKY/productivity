/* ============================================================
   04_city_sell_sharded —— 表4 city_sell：按源表分 5 片（方案2）
   背景：共享服务器，C盘(tempdb) 与 D盘(三个库) 均已 0 GB。
        不动配置、不建任何永久表。把原来「5 张源表 UNION 后一次聚合」
        改成「每张源表各聚合一次」，共 5 次查询、各导一个 CSV。
        每次排序量约为原来的 1/5，且没有 staging 写入磁盘。
   导出目录：G:\Kuangyu_Temp\Outsource\productivity\productivity\large_sample\city_sell_parts\
   ============================================================ */

/* 确认空间已释放（空闲MB 应从 5 万明显变大） */
USE GX17;
SELECT name AS 逻辑名,
       CAST(size/128.0 AS decimal(18,1))                                    AS 文件MB,
       CAST(FILEPROPERTY(name,'SpaceUsed')/128.0 AS decimal(18,1))          AS 已用MB,
       CAST((size - FILEPROPERTY(name,'SpaceUsed'))/128.0 AS decimal(18,1)) AS 空闲MB
FROM sys.database_files;


/* ############################################################
   5 片查询：每张源表跑一次，各导一个 CSV。
   注意：同一 (地区,产品,单位,年) 可能在多片里都出现 —— 这是正常的，
        Python 端会把各片按键合并：金额/数量相加。
   ★ 卖方企业数不能直接相加（会重复计数），见文末说明。
   ############################################################ */

/* --- 片1：GX1701（2017）→ city_sell_p1_1701.csv --- */
SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数,
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX17.dbo.GX1701
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 片2：GX1702（2017）→ city_sell_p2_1702.csv --- */
SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数,
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX17.dbo.GX1702
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 片3：GX1703（2017）→ city_sell_p3_1703.csv --- */
SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数,
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX17.dbo.GX1703
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 片4：gaoxiao1801（2018上半年）→ city_sell_p4_1801.csv --- */
SELECT 销方地区, 项目代码, 单位, 2018 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数,
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX.dbo.gaoxiao1801
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 片5：GX02（2018下半年）→ city_sell_p5_GX02.csv --- */
SELECT 销方地区, 项目代码, 单位, 2018 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数,
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX02.dbo.GX02
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;


/* ############################################################
   ★ 关于「卖方企业数」的口径说明（重要）
   分片后每片只能数出「该片内的不同卖方数」，跨片相加会重复计数
   （同一家企业在 1701 和 1702 都卖过，会被数两次）。
   处理方式二选一：
   (a) 先用上面 5 片把 金额/数量 拿到（这两个相加是精确的）；
       卖方企业数另外单独跑下面这条「只查企业数」的轻查询（列少、
       不带金额数量，排序压力小得多），一次出准确值：
   (b) 若 (a) 仍跑不动，再按年拆成 2017/2018 两次跑。

   --- 只查卖方企业数（2017）→ city_sell_nseller_2017.csv ---
   SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
          COUNT(DISTINCT 销方企业ID) AS 卖方企业数
   FROM (
       SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX17.dbo.GX1701
       UNION ALL SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX17.dbo.GX1702
       UNION ALL SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX17.dbo.GX1703
   ) a
   WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
   GROUP BY 销方地区, 项目代码, 单位;

   --- 只查卖方企业数（2018）→ city_sell_nseller_2018.csv ---
   SELECT 销方地区, 项目代码, 单位, 2018 AS [year],
          COUNT(DISTINCT 销方企业ID) AS 卖方企业数
   FROM (
       SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX.dbo.gaoxiao1801
       UNION ALL SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX02.dbo.GX02
   ) a
   WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
   GROUP BY 销方地区, 项目代码, 单位;
   ############################################################ */
