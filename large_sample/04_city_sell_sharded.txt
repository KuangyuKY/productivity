/* ============================================================
   04_city_sell_sharded —— 表4 city_sell（分片版）
   为什么单独一个文件：表4（全量卖方侧）数据量太大，01 里那条
   「5 张源表 UNION 后一次聚合」跑不动（tempdb / 磁盘吃紧）。
   这里拆成 5 片，每片只扫一张源表，跑完各导一个 CSV。
   不建任何 staging 表，不往数据库写数据。

   导出目录：G:\Kuangyu_Temp\Outsource\productivity\productivity\large_sample\city_sell_parts\
   ============================================================ */


/* ############################################################
   A. 金额 / 数量 —— 5 片，逐片跑、逐片导出
      这两个量跨片相加是精确的，Python 端按
      (销方地区, 项目代码, 单位, year) 合并求和。
   ############################################################ */

/* --- 片1：GX1701（2017）→ city_sell_p1.csv --- */
SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX17.dbo.GX1701
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 片2：GX1702（2017）→ city_sell_p2.csv --- */
SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX17.dbo.GX1702
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 片3：GX1703（2017）→ city_sell_p3.csv --- */
SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX17.dbo.GX1703
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 片4：gaoxiao1801（2018上半年）→ city_sell_p4.csv --- */
SELECT 销方地区, 项目代码, 单位, 2018 AS [year],
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX.dbo.gaoxiao1801
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 片5：GX02（2018下半年）→ city_sell_p5.csv --- */
SELECT 销方地区, 项目代码, 单位, 2018 AS [year],
       SUM(开票金额)               AS 金额合计,
       SUM(TRY_CAST(数量 AS float)) AS 数量合计
FROM GX02.dbo.GX02
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;


/* ############################################################
   B. 卖方企业数 —— 必须单独跑，不能用 A 的分片相加
      原因：COUNT(DISTINCT) 不可加。同一家企业在 1701 和 1702 都卖过，
      分片各数一次，相加会重复计数。
      所以按年各跑一条：同一年内部先 UNION ALL 再去重，
      与 01 里原始写法的口径完全一致。
      这两条只 SELECT 四列、不带金额数量，比完整查询轻很多。
   ############################################################ */

/* --- 2017 卖方企业数 → city_sell_nseller_2017.csv --- */
SELECT 销方地区, 项目代码, 单位, 2017 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数
FROM (
    SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX17.dbo.GX1701
    UNION ALL SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX17.dbo.GX1702
    UNION ALL SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX17.dbo.GX1703
) a
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;

/* --- 2018 卖方企业数 → city_sell_nseller_2018.csv --- */
SELECT 销方地区, 项目代码, 单位, 2018 AS [year],
       COUNT(DISTINCT 销方企业ID) AS 卖方企业数
FROM (
    SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX.dbo.gaoxiao1801
    UNION ALL SELECT 销方地区, 销方企业ID, 项目代码, 单位 FROM GX02.dbo.GX02
) a
WHERE 销方地区 IS NOT NULL AND 项目代码 IS NOT NULL
GROUP BY 销方地区, 项目代码, 单位;
